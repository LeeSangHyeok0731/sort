```ts
import { resolveTheme, THEME_INIT_SCRIPT } from "./theme";

describe("resolveTheme", () => {
  it("저장된 값이 있으면 OS 설정보다 우선한다", () => {
    expect(resolveTheme("light", true)).toBe("light");
    expect(resolveTheme("dark", false)).toBe("dark");
  });

  it("저장된 값이 없으면 OS 설정을 따른다", () => {
    expect(resolveTheme(null, true)).toBe("dark");
    expect(resolveTheme(null, false)).toBe("light");
  });

  it("알 수 없는 저장 값은 무시한다", () => {
    expect(resolveTheme("blue", true)).toBe("dark");
    expect(resolveTheme("", false)).toBe("light");
  });
});

describe("THEME_INIT_SCRIPT", () => {
  /** 스크립트를 가짜 브라우저 전역으로 실행하고 정해진 data-theme를 돌려준다 */
  function run(stored: string | null, prefersDark: boolean, storageThrows = false): string | undefined {
    const root = { dataset: {} as Record<string, string> };
    const localStorage = {
      getItem: () => {
        if (storageThrows) throw new Error("blocked");
        return stored;
      },
    };
    const matchMedia = () => ({ matches: prefersDark });
    new Function("localStorage", "matchMedia", "document", THEME_INIT_SCRIPT)(
      localStorage,
      matchMedia,
      { documentElement: root }
    );
    return root.dataset.theme;
  }

  it.each([
    ["light", true],
    ["dark", false],
    [null, true],
    [null, false],
    ["blue", true],
  ] as const)("저장 %s, OS 다크 %s → resolveTheme와 같은 결과", (stored, prefersDark) => {
    expect(run(stored, prefersDark)).toBe(resolveTheme(stored, prefersDark));
  });

  it("localStorage가 막혀 있어도 OS 설정으로 정한다", () => {
    expect(run(null, true, true)).toBe("dark");
  });
});
```

```ts
export type Theme = "light" | "dark";

/** localStorage 키. 값은 "light" | "dark" */
export const THEME_STORAGE_KEY = "theme";

/** 저장된 선택이 있으면 그것을, 없으면 OS 설정을 따른다 */
export function resolveTheme(stored: string | null, prefersDark: boolean): Theme {
  if (stored === "light" || stored === "dark") return stored;
  return prefersDark ? "dark" : "light";
}

/** 지금 화면에 적용된 테마 (<html data-theme>) */
export function readTheme(): Theme {
  return document.documentElement.dataset.theme === "dark" ? "dark" : "light";
}

/** 테마를 바꾸고 선택을 기억한다 */
export function setTheme(theme: Theme) {
  document.documentElement.dataset.theme = theme;
  try {
    localStorage.setItem(THEME_STORAGE_KEY, theme);
  } catch {
    // 저장이 막혀 있으면 이번 방문 동안만 적용
  }
}

/**
 * <head>에서 화면을 그리기 전에 실행해 밝은 화면이 번쩍이지 않게 한다.
 * resolveTheme와 같은 규칙이어야 한다 (theme.test.ts가 확인).
 */
export const THEME_INIT_SCRIPT = `(function(){var s=null;try{s=localStorage.getItem("${THEME_STORAGE_KEY}")}catch(e){}var d=s==="dark"||(s!=="light"&&matchMedia("(prefers-color-scheme: dark)").matches);document.documentElement.dataset.theme=d?"dark":"light"})()`;
```

`src/shared/lib/theme/useTheme.ts` (`"use client"` 지시어는 넣지 않는다. layout.tsx가 같은 묶음에서 문자열 상수를 가져가기 때문):

```ts
import { useSyncExternalStore } from "react";
import { readTheme, type Theme } from "./theme";

function subscribe(onChange: () => void) {
  const observer = new MutationObserver(onChange);
  observer.observe(document.documentElement, { attributes: true, attributeFilter: ["data-theme"] });
  return () => observer.disconnect();
}

/** 지금 테마. 서버 렌더와 첫 hydration에서는 "light"이고, 곧 실제 값으로 다시 그린다 */
export function useTheme(): Theme {
  return useSyncExternalStore(subscribe, readTheme, () => "light");
}
```

```ts
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";
import { describe, expect, it } from "vitest";
import { PALETTE } from "./palette";

const css = readFileSync(fileURLToPath(new URL("../../app/globals.css", import.meta.url)), "utf8");

/** CSS 블록 본문 ({ 다음부터 짝이 맞는 } 앞까지) */
function block(selector: string): string {
  const start = css.indexOf(`${selector} {`);
  if (start < 0) return "";
  const open = css.indexOf("{", start);
  return css.slice(open + 1, css.indexOf("}", open));
}

const BLOCKS = {
  light: block("@theme"),
  dark: block(':root[data-theme="dark"]'),
} as const;

function cssToken(source: string, name: string): string | undefined {
  return source.match(new RegExp(`--color-${name}:\\s*(#[0-9a-fA-F]{6})\\s*;`))?.[1].toLowerCase();
}

describe.each(["light", "dark"] as const)("PALETTE.%s", (theme) => {
  const p = PALETTE[theme];
  const flat: Array<[string, string]> = [
    ["background", p.background],
    ["surface", p.surface],
    ["subtle", p.subtle],
    ["line", p.line],
    ["line-strong", p.lineStrong],
    ["foreground", p.foreground],
    ["muted", p.muted],
    ["faint", p.faint],
    ...Object.entries(p.tone).map(([k, v]) => [`tone-${k}`, v] as [string, string]),
    ["slot-1", p.slot[0]],
    ["slot-2", p.slot[1]],
  ];

  it("CSS 블록을 찾는다", () => {
    expect(BLOCKS[theme]).not.toBe("");
  });

  it.each(flat)("--color-%s 가 globals.css와 같다", (name, value) => {
    expect(cssToken(BLOCKS[theme], name)).toBe(value.toLowerCase());
  });
});
```

- [ ] **Step 2: 버튼** — `src/features/toggle-theme/ui/ThemeToggle.tsx`

```tsx
"use client";

import { cn, setTheme, useTheme } from "@/shared/lib";
import ThemeIcon from "./ThemeIcon";

export default function ThemeToggle({ className }: { className?: string }) {
  const dark = useTheme() === "dark";
  const label = dark ? "밝은 테마로" : "어두운 테마로";
  return (
    <button
      type="button"
      onClick={() => setTheme(dark ? "light" : "dark")}
      aria-label={label}
      title={label}
      className={cn(
        "w-9 h-9 flex items-center justify-center rounded-lg transition-all active:scale-95 focus-visible:outline-2 focus-visible:outline-foreground",
        "text-muted hover:text-foreground hover:bg-subtle",
        className
      )}
    >
      <ThemeIcon dark={dark} />
    </button>
  );
}
```
